## 1. Documentation

### Overview

- Object storage organised as **buckets** that hold **keys** (objects).
- In this project four buckets are used: `raw` (landing, versioned, 30-day lifecycle), `curated` (cleaned output), `scripts` (Glue scripts) and `athena-results`.
- Bucket notifications turn uploads into events for SQS or Lambda.

### At a glance

| Item | Value |
|------|-------|
| boto3 client | `s3` |
| Terraform endpoint | `s3` |
| Needs first | nothing (notifications need the SQS/Lambda destination to exist) |
| Docker socket | not required |

### How it behaves on Floci

- Use **path-style** URLs (`http://localhost:4566/<bucket>/<key>`). The shared client already sets this.
- `put_bucket_notification_configuration` **checks that the SQS/SNS/Lambda destination exists**, and sends an `s3:TestEvent` message to SQS and SNS destinations when you configure it. That message has no `Records` array, so consumers must ignore it (the ingest Lambda does).
- Subscribe with `s3:ObjectCreated:*`. Browser/presigned-POST uploads emit `ObjectCreated:Post`, not `Put`.
- Destination resource-policy permissions are not checked by Floci, but add them anyway so the same code works on real AWS.

### Key concepts

- **Versioning**: keeps every object version; deleting a versioned bucket needs version-aware deletion.
- **Lifecycle rule**: expires objects under a prefix after N days.
- **Notification filter**: limits events to keys with a given prefix and/or suffix.
- **Presigned URL**: a time-limited link to one object.

### Gotchas

- `us-east-1` must not send `CreateBucketConfiguration`; every other region must. `create_bucket` in Example 1 handles both.
- Versioned buckets need version-aware deletion (`empty_and_delete`) or Terraform `force_destroy = true`.
- Notification filters are a prefix/suffix pair; `raw/` + `.csv` only fires for CSV uploads under `raw/`.

## 2. Basic Usage

> Prerequisite: Floci is running and the `AWS_*` environment variables are set ([guide 00](00-setup-floci.md)).

```python
import boto3
from botocore.config import Config

s3 = boto3.client("s3", config=Config(s3={"addressing_style": "path"}))  # path-style on Floci

s3.create_bucket(Bucket="demo-bucket")  # us-east-1: no LocationConstraint
s3.put_object(Bucket="demo-bucket", Key="raw/hello.txt", Body=b"hello floci")

body = s3.get_object(Bucket="demo-bucket", Key="raw/hello.txt")["Body"].read().decode()
print(body)                                                    # hello floci

keys = [o["Key"] for o in s3.list_objects_v2(Bucket="demo-bucket", Prefix="raw/").get("Contents", [])]
print(keys)                                                    # ['raw/hello.txt']

s3.delete_object(Bucket="demo-bucket", Key="raw/hello.txt")
s3.delete_bucket(Bucket="demo-bucket")
```

## 3. Examples

### Example 1 — Pipeline buckets with versioning and lifecycle

Creates `raw`, `curated`, `scripts` and `athena-results`, versions `raw`, and expires `raw/` after 30 days.

`src/s3_ops/buckets.py`

```python
from botocore.exceptions import ClientError

from src.common.aws import REGION, client, name

s3 = client("s3")

PIPELINE_BUCKETS = ("raw", "curated", "scripts", "athena-results")


def create_bucket(bucket: str, versioning: bool = False) -> str:
    kwargs = {"Bucket": bucket}
    if REGION != "us-east-1":  # us-east-1 must NOT send a LocationConstraint
        kwargs["CreateBucketConfiguration"] = {"LocationConstraint": REGION}
    try:
        s3.create_bucket(**kwargs)
    except ClientError as e:
        if e.response["Error"]["Code"] not in ("BucketAlreadyOwnedByYou", "BucketAlreadyExists"):
            raise
    if versioning:
        s3.put_bucket_versioning(Bucket=bucket, VersioningConfiguration={"Status": "Enabled"})
    return bucket


def set_lifecycle(bucket: str, prefix: str = "raw/", days: int = 30) -> None:
    s3.put_bucket_lifecycle_configuration(
        Bucket=bucket,
        LifecycleConfiguration={"Rules": [{
            "ID": f"expire-{prefix.strip('/').replace('/', '-')}",
            "Status": "Enabled",
            "Filter": {"Prefix": prefix},
            "Expiration": {"Days": days},
        }]},
    )


def create_pipeline_buckets() -> dict[str, str]:
    out = {}
    for suffix in PIPELINE_BUCKETS:
        out[suffix] = create_bucket(name(suffix), versioning=(suffix == "raw"))
    set_lifecycle(out["raw"])
    return out


def empty_and_delete(bucket: str) -> None:
    """Delete every object version and delete marker, then the bucket."""
    for page in s3.get_paginator("list_object_versions").paginate(Bucket=bucket):
        objs = [{"Key": v["Key"], "VersionId": v["VersionId"]}
                for v in page.get("Versions", []) + page.get("DeleteMarkers", [])]
        if objs:
            s3.delete_objects(Bucket=bucket, Delete={"Objects": objs})
    s3.delete_bucket(Bucket=bucket)


def list_buckets() -> list[str]:
    return [b["Name"] for b in s3.list_buckets()["Buckets"]]


if __name__ == "__main__":
    print(create_pipeline_buckets())
    print(list_buckets())
```

```bash
python -m src.s3_ops.buckets
```

### Example 2 — Object helpers (text, JSON, upload, download, presign, copy)

`src/s3_ops/objects.py`

```python
import json
from pathlib import Path

from src.common.aws import client

s3 = client("s3")


def upload_file(bucket: str, key: str, path: str, content_type: str | None = None) -> None:
    extra = {"ContentType": content_type} if content_type else None
    s3.upload_file(path, bucket, key, ExtraArgs=extra)


def put_text(bucket: str, key: str, text: str, content_type: str = "text/plain") -> None:
    s3.put_object(Bucket=bucket, Key=key, Body=text.encode(), ContentType=content_type)


def get_text(bucket: str, key: str) -> str:
    return s3.get_object(Bucket=bucket, Key=key)["Body"].read().decode()


def put_json(bucket: str, key: str, payload: dict) -> None:
    put_text(bucket, key, json.dumps(payload), "application/json")


def get_json(bucket: str, key: str) -> dict:
    return json.loads(get_text(bucket, key))


def download(bucket: str, key: str, dest: str) -> Path:
    Path(dest).parent.mkdir(parents=True, exist_ok=True)
    s3.download_file(bucket, key, dest)
    return Path(dest)


def list_keys(bucket: str, prefix: str = "") -> list[str]:
    keys: list[str] = []
    for page in s3.get_paginator("list_objects_v2").paginate(Bucket=bucket, Prefix=prefix):
        keys += [o["Key"] for o in page.get("Contents", [])]
    return keys


def presigned_get(bucket: str, key: str, expires: int = 900) -> str:
    return s3.generate_presigned_url("get_object", Params={"Bucket": bucket, "Key": key},
                                     ExpiresIn=expires)


def copy(src_bucket: str, src_key: str, dst_bucket: str, dst_key: str) -> None:
    s3.copy_object(Bucket=dst_bucket, Key=dst_key,
                   CopySource={"Bucket": src_bucket, "Key": src_key})


def delete(bucket: str, key: str) -> None:
    s3.delete_object(Bucket=bucket, Key=key)
```

### Example 3 — Event notifications to SQS or Lambda

The destination must already exist (Floci checks this).

`src/s3_ops/notifications.py`

```python
"""Bucket event notifications. The destination must already exist (Floci checks this)."""
from src.common.aws import client

s3 = client("s3")


def _filter(prefix: str, suffix: str) -> dict:
    rules = []
    if prefix:
        rules.append({"Name": "prefix", "Value": prefix})
    if suffix:
        rules.append({"Name": "suffix", "Value": suffix})
    return {"Key": {"FilterRules": rules}} if rules else {}


def notify_sqs(bucket: str, queue_arn: str, prefix: str = "raw/", suffix: str = ".csv") -> None:
    s3.put_bucket_notification_configuration(
        Bucket=bucket,
        NotificationConfiguration={"QueueConfigurations": [{
            "QueueArn": queue_arn,
            "Events": ["s3:ObjectCreated:*"],
            "Filter": _filter(prefix, suffix),
        }]},
    )


def notify_lambda(bucket: str, function_arn: str, prefix: str = "raw/", suffix: str = "") -> None:
    s3.put_bucket_notification_configuration(
        Bucket=bucket,
        NotificationConfiguration={"LambdaFunctionConfigurations": [{
            "LambdaFunctionArn": function_arn,
            "Events": ["s3:ObjectCreated:*"],
            "Filter": _filter(prefix, suffix),
        }]},
    )


def clear(bucket: str) -> None:
    s3.put_bucket_notification_configuration(Bucket=bucket, NotificationConfiguration={})
```

### Example 4 — Sample data (used by the Glue and Athena guides)

A small `orders.csv` with a duplicate and an invalid row on purpose.

`data/make_orders.py`

```python
"""Generate a small orders.csv (includes a duplicate and a bad amount on purpose)."""
import csv
import random
from pathlib import Path


def make(path: str = "data/orders.csv", n: int = 50, seed: int = 7) -> str:
    random.seed(seed)
    rows = [{"order_id": f"O-{i:04d}", "customer_id": f"C-{random.randint(1, 8)}",
             "amount": round(random.uniform(5, 500), 2), "status": "NEW"} for i in range(1, n + 1)]
    rows.append(dict(rows[0]))                                           # duplicate order_id
    rows.append({"order_id": "O-9999", "customer_id": "C-1", "amount": -5.0, "status": "NEW"})  # invalid
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w", newline="") as f:
        w = csv.DictWriter(f, fieldnames=["order_id", "customer_id", "amount", "status"])
        w.writeheader()
        w.writerows(rows)
    return path


if __name__ == "__main__":
    print(make())
```

### Example 5 — Test

`tests/test_01_iam_s3.py`

```python
from src.common.aws import name
from src.iam_ops import roles
from src.s3_ops import buckets, objects


def test_iam_roles_roundtrip():
    arns = roles.create_all()
    assert set(arns) == set(roles.ROLES)
    assert roles.create_all() == arns  # idempotent
    for suffix in ("lambda-role", "glue-role"):
        assert arns[suffix].endswith(name(suffix))


def test_s3_buckets_and_objects(tmp_path):
    b = buckets.create_pipeline_buckets()
    raw = b["raw"]
    assert raw in buckets.list_buckets()

    objects.put_json(raw, "raw/a.json", {"x": 1})
    assert objects.get_json(raw, "raw/a.json") == {"x": 1}
    f = tmp_path / "t.csv"
    f.write_text("a,b\n1,2\n")
    objects.upload_file(raw, "raw/t.csv", str(f))
    assert sorted(objects.list_keys(raw, "raw/")) == ["raw/a.json", "raw/t.csv"]
    objects.copy(raw, "raw/t.csv", b["curated"], "copy/t.csv")
    assert objects.get_text(b["curated"], "copy/t.csv") == "a,b\n1,2\n"
    assert objects.presigned_get(raw, "raw/t.csv").startswith("http")
    out = objects.download(raw, "raw/t.csv", str(tmp_path / "dl" / "t.csv"))
    assert out.read_text() == "a,b\n1,2\n"
```

### Example 6 — Terraform

`terraform/s3.tf`

```hcl
resource "aws_s3_bucket" "this" {
  for_each      = toset(["raw", "curated", "scripts", "athena-results"])
  bucket        = "${var.prefix}-${each.key}"
  force_destroy = true
  tags          = local.tags
}

resource "aws_s3_bucket_versioning" "raw" {
  bucket = aws_s3_bucket.this["raw"].id
  versioning_configuration {
    status = "Enabled"
  }
}

resource "aws_s3_bucket_lifecycle_configuration" "raw" {
  bucket = aws_s3_bucket.this["raw"].id
  rule {
    id     = "expire-raw"
    status = "Enabled"
    filter {
      prefix = "raw/"
    }
    expiration {
      days = 30
    }
  }
  depends_on = [aws_s3_bucket_versioning.raw]
}

# S3 -> SQS notification (queue policy lives in sqs.tf)
resource "aws_s3_bucket_notification" "raw" {
  bucket = aws_s3_bucket.this["raw"].id

  queue {
    queue_arn     = aws_sqs_queue.main.arn
    events        = ["s3:ObjectCreated:*"]
    filter_prefix = "raw/"
    filter_suffix = ".csv"
  }

  depends_on = [aws_sqs_queue_policy.main]
}
```
